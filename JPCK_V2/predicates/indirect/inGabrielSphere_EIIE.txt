qx qy qz cx cy cz

bax = l3x * d2 - l2x * d3
bay = l3y * d2 - l2y * d3
baz = l3z * d2 - l2z * d3
cax = cx * d2 - l2x 
cay = cy * d2 - l2y 
caz = cz * d2 - l2z
qax = qx * d2 - l2x 
qay = qy * d2 - l2y 
qaz = qz * d2 - l2z

ba2 = bax^2 + bay^2 + baz^2 // d2^2 d3^2
ca2 = (cax^2 + cay^2 + caz^2) * d3 // d2^2 d3

abcx = cax * ba2 - bax * ca2 // d2^3 d3^2
abcy = cay * ba2 - bay * ca2 
abcz = caz * ba2 - baz * ca2

crossbcx = bay * caz - baz * cay // d2^2 d3
crossbcy = baz * cax - bax * caz
crossbcz = bax * cay - bay * cax

ccax = abcy * crossbcz - abcz * crossbcy // d2^5 d3^3
ccay = abcz * crossbcx - abcx * crossbcz
ccaz = abcx * crossbcy - abcy * crossbcx

c2 = (crossbcx^2 + crossbcy^2 + crossbcz^2) * d3 // d2^4 d3^2
ret = qax * (qax * c2 - ccax) + qay * (qay * c2 - ccay) + qaz * (qaz * c2 - ccaz)
