genericPoint(a:l2x,l2y,l2z,d2)
qx qy qz bx by bz cx cy cz

bax = bx * d2 - l2x 
bay = by * d2 - l2y 
baz = bz * d2 - l2z
cax = cx * d2 - l2x 
cay = cy * d2 - l2y 
caz = cz * d2 - l2z
qax = qx * d2 - l2x 
qay = qy * d2 - l2y 
qaz = qz * d2 - l2z

ba2 = bax^2 + bay^2 + baz^2 // d2^2
ca2 = cax^2 + cay^2 + caz^2

abcx = cax * ba2 - bax * ca2 // d2^3
abcy = cay * ba2 - bay * ca2 
abcz = caz * ba2 - baz * ca2

crossbcx = bay * caz - baz * cay // d2^2
crossbcy = baz * cax - bax * caz
crossbcz = bax * cay - bay * cax

ccax = abcy * crossbcz - abcz * crossbcy // d2^5
ccay = abcz * crossbcx - abcx * crossbcz
ccaz = abcx * crossbcy - abcy * crossbcx

c2 = (crossbcx^2 + crossbcy^2 + crossbcz^2)
ret = qax * (qax * c2 - ccax) + qay * (qay * c2 - ccay) + qaz * (qaz * c2 - ccaz)
